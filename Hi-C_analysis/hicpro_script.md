#!/bin/bash
################################################################################
# Hi-C Data Analysis Pipeline using HiC-Pro
#
# Description:
#   Complete pipeline for processing Hi-C sequencing data from raw FASTQ files
#   to analyzed contact matrices with HiC-Pro, and converts output to Juicebox format
#   for visualization and downstream analysis.
#
# Author: Jothivanan Elumalai
# License: MIT
# Version: 3.1.0
# Date: 2025
#
# Citation:
#   If you use this pipeline, please cite:
#   - Servant et al., Genome Biology 2015 (HiC-Pro)
#   - Elumalai and Hiratani, in revision
#
# Requirements:
#   - Conda environment with HiC-Pro installed
#   - Juicer tools JAR file
#   - Bowtie2 indexed reference genome
#   - tmux (optional)
#   - Chromosome sizes file (Example: hg19.chrom.sizes)
#   - Restriction enzyme sites (generated by digest_genome.py script in HiC-Pro)
#   - HiC-Pro configuration file
#
# Pipeline Steps:
#   Step 1: Generate restriction fragment file
#   Step 2: Edit HiC-Pro config file
#   Step 3: Run HiC-Pro analysis
#   Step 4: Convert to Juicebox format

################################################################################

set -e  # Exit on error
set -u  # Exit on undefined variable

## ---------------------------------------------------------------
## Configuration - EDIT THESE SETTINGS
## ---------------------------------------------------------------

# Project settings
Project_dir="${HOME}/HiC_project"          # Project directory
sample_name="Sample"                        # Sample name for outputs

# Reference genome files (UPDATE THESE PATHS)
genome_name="hg19"                                           # Genome build
genome_fasta="path/to/reference_genome/hg19.fa"             # Reference FASTA
chrom_sizes="path/to/reference_genome/hg19.chrom.sizes"     # Chromosome sizes

# Restriction enzyme
# Options: mboi, dpnii, hindiii etc.
restriction_enzyme="dpnii"                                   # 4-cutter enzyme

# HiC-Pro settings
hicpro_dir="${HOME}/HiC-Pro_3.1.0"                      # HiC-Pro installation directory
config_file="${Project_dir}/config.txt"               # HiC-Pro config file
# Example config file provided

# Juicer settings
juicer_tools_jar="path/to/juicer_tools.jar"                 # Juicer tools JAR

# Computational resources
threads=20              # Number of threads for processing

# Conda environment
conda_env="hicpro"      # Conda environment name

## ---------------------------------------------------------------
## Step 1: Generate Restriction Fragment File
## ---------------------------------------------------------------

echo "Step 1: Generating Restriction Fragment File"
echo "-----------------------------------------------------------"
echo "Restriction enzyme: ${restriction_enzyme}"
echo "Reference genome: ${genome_fasta}"
echo ""

conda activate ${conda_env}
resfrag_file="${HOME}/HiC-Pro_3.1.0/annotation/${restriction_enzyme}_resfrag_${genome_name}.bed"

if [ ! -f "${resfrag_file}" ]; then
    echo "Digesting genome..."
    digest_genome.py -r "${restriction_enzyme}" \
        -o "${resfrag_file}" \
        "${genome_fasta}"
    echo "✓ Restriction fragment file created"
else
    echo "✓ Restriction fragment file already exists"
fi

## ---------------------------------------------------------------
## Step 2: Edit HiC-Pro config file
## ---------------------------------------------------------------

# Example config file is given
# change inputs on lines where (# change here) is mentioned

## ---------------------------------------------------------------
## Step 3: Run HiC-Pro Analysis
## ---------------------------------------------------------------

echo "Step 3: Running HiC-Pro Analysis"
echo "-----------------------------------------------------------"
echo "Input: ${Project_dir}/fastq"
echo "Output: ${Project_dir}/hicpro"
echo "Config: ${config_file}"
echo ""
echo "This step may take several hours to complete."
echo ""

conda activate ${conda_env}
HiC-Pro -i "${Project_dir}/fastq" \
        -o "${Project_dir}/hicpro" \
        -c "${config_file}"
echo "✓ HiC-Pro analysis complete"
echo ""

## ---------------------------------------------------------------
## Step 4: Convert to Juicebox Format
## ---------------------------------------------------------------

echo "Step 4: Converting to Juicebox Format"
echo "-----------------------------------------------------------"

# Find allValidPairs file
valid_pairs=$(find "${Project_dir}/hicpro/hic_results/data" -name "*.allValidPairs" | head -n 1)

if [ -z "$valid_pairs" ]; then
    echo "ERROR: No allValidPairs file found!"
    echo "Check HiC-Pro output in: ${Project_dir}/hicpro/hic_results/data"
    exit 1
fi

echo "Input: ${valid_pairs}"
echo "Output: ${Project_dir}/juicebox"
echo "Juicer tools: ${juicer_tools_jar}"
echo ""

# Convert to Juicebox format
conda activate ${conda_env}
hicpro2juicebox.sh \
    -i "${valid_pairs}" \
    -g "${chrom_sizes}" \
    -o "${Project_dir}/juicebox" \
    -j "${juicer_tools_jar}"

echo "✓ Conversion to Juicebox format complete"
echo ""

## ---------------------------------------------------------------
## END
## ---------------------------------------------------------------
